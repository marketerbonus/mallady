<script>
  function toggleHint(step) {
    const hintBox = document.getElementById(`hint${step}`);
    hintBox.style.display = (hintBox.style.display === "block") ? "none" : "block";
  }

  function checkAnswer(step, target) {
    const input = document.getElementById(`ans${step}`).value.trim().toUpperCase();
    const feedback = document.getElementById(`fb${step}`);
    
    let isCorrect = false;

    if (Array.isArray(target)) {
      isCorrect = target.some(t => input.includes(t.toUpperCase()));
    } else {
      isCorrect = (input === target.toUpperCase());
    }

    if (isCorrect) {
      feedback.textContent = "✔ 정답입니다! 새로운 단서가 승인되었습니다.";
      feedback.className = "feedback correct";
      setTimeout(() => {
        document.getElementById(`p${step}`).classList.remove('active');
        const nextCard = document.getElementById(`p${step + 1}`) || document.getElementById('p-final');
        nextCard.classList.add('active');
      }, 800);
    } else {
      feedback.textContent = "✖ 단서 검증 실패. 다시 분석해보세요.";
      feedback.className = "feedback incorrect";
    }
  }
</script>
